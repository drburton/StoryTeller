COMMENT "# prologue.tale" [1]
COMMENT "# The first morning at school." [2]
ANNOTATION @title("Prologue") [3]
BLANK [4]
VAR trust := 0 [5]
VAR @global prologue_done := false [6]
BLANK [7]
BEAT start [8]
  EXPRESSION (call backdrop "classroom_morning" transition="fade" time=1.0) [9]
  EXPRESSION (call music "quiet_morning" volume=0.8) [10]
  EXPRESSION (call (. mira enter) "smile" at=LEFT) [11]
  BLANK [12]
  NARRATION "Sunlight spills across rows of empty desks." [13]
  DIALOGUE mira "You\'re here early.[pause] Couldn\'t sleep?" [14]
  DIALOGUE mira (curious) "Nervous about the exam?" [15]
  BLANK [16]
  CHOOSE [17]
    OPTION "Admit you were nervous" [18]
      ASSIGN trust += 1 [19]
      JUMP honest [20]
    OPTION "Act like everything is fine" [21]
      JUMP bluff [22]
    OPTION "Change the subject" if (> trust 2) [23]
      DIALOGUE mira "Smooth. Really smooth." [24]
BLANK [25]
BEAT honest [26]
  DIALOGUE mira (soft) "Same here, honestly." [27]
  EXPRESSION (await (call (. mira move_to) CENTER time=0.6)) [28]
  DIALOGUE mira "Want to study together at lunch?" [29]
  JUMP walk [30]
BLANK [31]
BEAT bluff [32]
  DIALOGUE mira (smirk) "Look at you, all confident." [33]
  JUMP walk [34]
BLANK [35]
BEAT walk [36]
  IF (> trust 0) [37]
    DIALOGUE mira "Let\'s head out, {player_name}." [38]
  ELSE [39]
    DIALOGUE mira "Well. See you in class." [40]
  BLANK [41]
  EXPRESSION (call (. mira exit) transition="slide_left") [42]
  ASSIGN prologue_done = true [43]
  JUMP chapter_1.start [44]
